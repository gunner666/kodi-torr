async function fetchRepoTree() { try { // === DETECCIÓN AUTOMÁTICA DE DATOS === // Esto lee automáticamente tu usuario y repositorio desde la URL de GitHub Pages const pathParts = window.location.pathname.split('/').filter(p => p !== ''); const GITHUB_USER = window.location.hostname.split('.')[0]; const GITHUB_REPO = pathParts[0] || ''; const BRANCH = 'main'; // Cambia por 'master' si tu rama se llama así // Si estás probando localmente en tu PC, usará estos datos por defecto const finalUser = GITHUB_USER === 'localhost' || GITHUB_USER === '127' ? 'ESCRIBE_AQUI_TU_USUARIO_REAL' : GITHUB_USER; const finalRepo = GITHUB_USER === 'localhost' || GITHUB_USER === '127' ? 'ESCRIBE_AQUI_TU_REPO_REAL' : GITHUB_REPO; const API_URL = `https://github.com{finalUser}/${finalRepo}/git/trees/${BRANCH}?recursive=1`; // Petición a la API const response = await fetch(API_URL).catch(() => { throw new Error("El navegador bloqueó la conexión física. Intenta desactivar AdBlockers/Extensiones de privacidad o usa el Modo Incógnito."); }); if (response.status === 404) { throw new Error(`GitHub devolvió un error 404. Revisa si la rama se llama '${BRANCH}' o si el repositorio es realmente público.`); } if (!response.ok) { throw new Error(`Error en el servidor de GitHub (Código: ${response.status})`); } const data = await response.json(); const loadingEl = document.getElementById('loading'); if(loadingEl) loadingEl.remove(); const treeStructure = buildTreeStructure(data.tree); renderTree(treeStructure, document.getElementById('file-tree'), finalUser, finalRepo, BRANCH); setupEventListeners(); } catch (error) { document.getElementById('file-tree').innerHTML = `
Error detectado:
${error.message}
`; } } // Convierte la lista plana de GitHub en un objeto anidado (árbol) function buildTreeStructure(files) { const root = {}; files.forEach(file => { const parts = file.path.split('/'); let current = root; parts.forEach((part, index) => { if (!current[part]) { current[part] = { _meta: index === parts.length - 1 ? { type: file.type, url: file.url, path: file.path } : { type: 'tree' } }; } current = current[part]; }); }); return root; } // Dibuja el HTML del árbol de forma recursiva function renderTree(node, container, user, repo, branch) { const ul = document.createElement('ul'); ul.className = "space-y-1"; const keys = Object.keys(node).filter(k => k !== '_meta').sort((a, b) => { const aType = node[a]._meta.type; const bType = node[b]._meta.type; if (aType === 'tree' && bType !== 'tree') return -1; if (aType !== 'tree' && bType === 'tree') return 1; return a.localeCompare(b); }); keys.forEach(key => { const item = node[key]; const li = document.createElement('li'); li.className = "py-1"; if (item._meta.type === 'tree') { li.innerHTML = ` ${key} `; const nestedUlContainer = document.createElement('div'); nestedUlContainer.className = "nested"; renderTree(item, nestedUlContainer, user, repo, branch); li.appendChild(nestedUlContainer); } else { // No mostrar el propio index.html en el árbol para mantenerlo limpio if (key === 'index.html') return; const rawUrl = `https://githubusercontent.com{user}/${repo}/${branch}/${item._meta.path}`; li.innerHTML = `
${key}
Ver Descargar
`; } ul.appendChild(li); }); container.appendChild(ul); } function setupEventListeners() { document.querySelectorAll('.caret').forEach(caret => { caret.addEventListener('click', function(e) { e.stopPropagation(); const nested = this.parentElement.querySelector('.nested'); if (nested) { nested.classList.toggle('active'); this.classList.toggle('caret-down'); const icon = this.querySelector('.fa-folder, .fa-folder-open'); if (icon) { icon.classList.toggle('fa-folder'); icon.classList.toggle('fa-folder-open'); } } }); }); } // Inicializar automáticamente fetchRepoTree();
